#version 330
#define MAX_LIGHTS 10
uniform vec3 viewPosition;
uniform vec3 lightPositions[MAX_LIGHTS];
uniform int numLights;
uniform vec3 spotLightPosition;
uniform vec3 spotLightDirection;
uniform float spotLightInnerCutOff;
uniform float spotLightOuterCutOff;
in vec4 ex_worldPosition;
in vec3 ex_worldNormal;
out vec4 out_Color;
void main(void) {
    vec4 ambient = vec4(0.1, 0.1, 0.1, 1.0);
    vec4 diffuseSum = vec4(0.0);
    vec4 specularSum = vec4(0.0);

    vec3 normal = normalize(ex_worldNormal);
    vec3 fragPosition = vec3(ex_worldPosition);
    vec3 viewDir = normalize(viewPosition - fragPosition);

    float constantAttenuation = 1.0;
    float linearAttenuation = 0.09;
    float quadraticAttenuation = 0.032;
    //spotLightPosition = vec3(0.0, 2.0, 0.0);
    //spotLightDirection = vec3(0.0, -1.0, 0.0);

    for (int i = 0; i < numLights; i++) {
        vec3 lightDir = lightPositions[i] - fragPosition;
        float distance = length(lightDir);
        vec3 lightVector = normalize(lightDir);
        float attenuation = 1.0 / (constantAttenuation + linearAttenuation * distance + quadraticAttenuation * (distance * distance));
        float diff = max(dot(lightVector, normal), 0.0);
        vec4 diffuse = diff * vec4(0.0, 0.5, 0.0, 1.0) * attenuation;
        diffuseSum += diffuse;
        vec3 reflectDir = reflect(-lightVector, normal);
        float spec = pow(max(dot(viewDir, reflectDir), 0.0), 32.0);
        vec4 specular = spec * vec4(1.0, 1.0, 1.0, 1.0) * attenuation;
        specularSum += specular;
    }

    vec3 spotDir = normalize(spotLightPosition - fragPosition);

    float theta = dot(spotDir, normalize(-spotLightDirection));
    float epsilon = cos(radians(spotLightInnerCutOff)) - cos(radians(spotLightOuterCutOff));
    float intensity = clamp((theta - cos(radians(spotLightOuterCutOff))) / epsilon, 0.0, 1.0);

    if (theta > cos(radians(spotLightOuterCutOff))) {
        float spotDistance = length(spotLightPosition - fragPosition);
        float spotAttenuation = 1.0 / (constantAttenuation + linearAttenuation * spotDistance + quadraticAttenuation * (spotDistance * spotDistance));

        float spotDiff = max(dot(spotDir, normal), 0.0);
        vec4 spotDiffuse = spotDiff * vec4(0.0, 0.5, 0.0, 1.0) * spotAttenuation * intensity;
        diffuseSum += spotDiffuse;

        vec3 spotReflectDir = reflect(-spotDir, normal);
        float spotSpec = pow(max(dot(viewDir, spotReflectDir), 0.0), 32.0);
        vec4 spotSpecular = spotSpec * vec4(1.0, 1.0, 1.0, 1.0) * spotAttenuation * intensity;
        specularSum += spotSpecular;
    }
    out_Color = ambient + diffuseSum + specularSum;
}
