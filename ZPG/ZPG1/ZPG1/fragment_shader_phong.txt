#version 330
#define MAX_LIGHTS 10
uniform vec3 viewPosition;
uniform vec3 lightPositions[MAX_LIGHTS];
uniform int numLights;
in vec4 ex_worldPosition;
in vec3 ex_worldNormal;
out vec4 out_Color;
void main(void) {
    vec4 ambient = vec4(0.1, 0.1, 0.1, 1.0);
    vec4 diffuseSum = vec4(0.0);
    vec4 specularSum = vec4(0.0);
    vec3 normal = normalize(ex_worldNormal);
    vec3 fragPosition = vec3(ex_worldPosition);
    vec3 viewDir = normalize(viewPosition - fragPosition);
    float constantAttenuation = 1.0;
    float linearAttenuation = 0.09;
    float quadraticAttenuation = 0.032;
    for (int i = 0; i < numLights; i++) {
        vec3 lightDir = lightPositions[i] - fragPosition;
        float distance = length(lightDir);
        vec3 lightVector = normalize(lightDir);
        float attenuation = 1.0 / (constantAttenuation + linearAttenuation * distance + quadraticAttenuation * (distance * distance));
        float diff = max(dot(lightVector, normal), 0.0);
        vec4 diffuse = diff * vec4(0.385, 0.647, 0.812, 1.0) * attenuation;
        diffuseSum += diffuse;
        vec3 reflectDir = reflect(-lightVector, normal);
        float spec = pow(max(dot(viewDir, reflectDir), 0.0), 32.0);
        vec4 specular = spec * vec4(1.0, 1.0, 1.0, 1.0) * attenuation;
        specularSum += specular;
    }
    out_Color = ambient + diffuseSum + specularSum;
}
