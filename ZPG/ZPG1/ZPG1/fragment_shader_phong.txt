#version 330
uniform vec3 viewPosition;
uniform vec3 lightPosition;
in vec4 ex_worldPosition;
in vec3 ex_worldNormal;
out vec4 out_Color;
void main(void) {
vec3 lightVector = normalize(lightPosition - vec3(ex_worldPosition));
vec3 viewDir = normalize(viewPosition - vec3(ex_worldPosition));
vec3 reflectDir = reflect(-lightVector, normalize(ex_worldNormal));
float diff = max(dot(lightVector, normalize(ex_worldNormal)), 0.0);
float spec = pow(max(dot(viewDir, reflectDir), 0.0), 32.0);
vec4 diffuse = diff * vec4(0.385, 0.647, 0.812, 1.0);
vec4 specular = spec * vec4(1.0, 1.0, 1.0, 1.0);
vec4 ambient = vec4(0.1, 0.1, 0.1, 1.0);
out_Color = ambient + diffuse + specular;
}
